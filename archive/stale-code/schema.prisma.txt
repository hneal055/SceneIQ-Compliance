// ============================================================
// SceneIQ-Compliance — schema.prisma
// Production Graph v1
// ============================================================

generator client {
  provider             = "prisma-client-py"
  interface            = "asyncio"
  recursive_type_depth = 5
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ============================================================
// EXISTING MODELS (unchanged)
// ============================================================

model Jurisdiction {
  id             String          @id @default(uuid())
  name           String
  code           String          @unique
  country        String
  type           String
  description    String?
  website        String?
  active         Boolean         @default(true)
  createdAt      DateTime        @default(now())
  updatedAt      DateTime        @updatedAt
  incentiveRules IncentiveRule[]
  productions    Production[]
  @@map("jurisdictions")
}

model IncentiveRule {
  id               String       @id @default(uuid())
  jurisdictionId   String
  jurisdiction     Jurisdiction @relation(fields: [jurisdictionId], references: [id], onDelete: Cascade)
  ruleName         String
  ruleCode         String       @unique
  incentiveType    String
  percentage       Float?
  fixedAmount      Float?
  minSpend         Float?
  maxCredit        Float?
  eligibleExpenses String[]
  excludedExpenses String[]
  effectiveDate    DateTime
  expirationDate   DateTime?
  requirements     String?
  active           Boolean      @default(true)
  createdAt        DateTime     @default(now())
  updatedAt        DateTime     @updatedAt
  @@index([jurisdictionId])
  @@map("incentive_rules")
}

model Expense {
  id             String     @id @default(uuid())
  productionId   String
  production     Production @relation(fields: [productionId], references: [id], onDelete: Cascade)
  category       String
  subcategory    String?
  description    String
  amount         Float
  expenseDate    DateTime
  paymentDate    DateTime?
  isQualifying   Boolean    @default(true)
  qualifyingNote String?
  vendorName     String?
  vendorLocation String?
  receiptNumber  String?
  invoiceNumber  String?
  createdAt      DateTime   @default(now())
  updatedAt      DateTime   @updatedAt
  @@index([productionId])
  @@index([expenseDate])
  @@map("expenses")
}

// ============================================================
// PRODUCTION — Central Graph Node
// Every other node connects here. This is the OS hub.
// ============================================================

model Production {
  id                String    @id @default(uuid())
  title             String
  productionType    String    // feature / tv_series / documentary / commercial
  jurisdictionId    String
  jurisdiction      Jurisdiction @relation(fields: [jurisdictionId], references: [id])
  budgetTotal       Float
  budgetQualifying  Float?
  startDate         DateTime
  endDate           DateTime?
  productionCompany String
  status            String    // planning / pre_production / production / post / completed
  contact           String?
  metadata          String?
  episodeCount      Int?      // TV series episode count

  // --- PRODUCTION GRAPH EDGES ---
  scenes            Scene[]
  schedules         ScheduleDay[]
  crewMembers       CrewMember[]
  locations         ProductionLocation[]
  budgetLines       BudgetLine[]
  signals           ProductionSignal[]
  expenses          Expense[]

  createdAt         DateTime  @default(now())
  updatedAt         DateTime  @updatedAt

  @@index([jurisdictionId])
  @@index([status])
  @@map("productions")
}

// ============================================================
// SCENE NODE
// Sourced from AURA screenplay analysis output
// ============================================================

model Scene {
  id           String     @id @default(uuid())
  productionId String
  production   Production @relation(fields: [productionId], references: [id], onDelete: Cascade)
  sceneNumber  String
  intExt       String     // INT / EXT
  location     String
  timeOfDay    String     // DAY / NIGHT / DUSK / DAWN
  pageCount    Float?
  characters   String[]   // character names in this scene
  shotCount    Int?
  vfxShots     Int?
  status       String     @default("unscheduled") // unscheduled / scheduled / shot / complete
  notes        String?
  createdAt    DateTime   @default(now())
  updatedAt    DateTime   @updatedAt
  @@index([productionId])
  @@index([status])
  @@map("scenes")
}

// ============================================================
// SCHEDULE DAY NODE
// One row per shoot day. scenes[] holds scene IDs for that day.
// ============================================================

model ScheduleDay {
  id           String     @id @default(uuid())
  productionId String
  production   Production @relation(fields: [productionId], references: [id], onDelete: Cascade)
  shootDay     Int        // Day 1, Day 2, etc.
  date         DateTime?
  sceneIds     String[]   // Scene IDs scheduled this day
  callTime     String?    // e.g. "07:00"
  wrapTime     String?
  locationId   String?
  notes        String?
  createdAt    DateTime   @default(now())
  updatedAt    DateTime   @updatedAt
  @@index([productionId])
  @@map("schedule_days")
}

// ============================================================
// CREW MEMBER NODE
// Feeds Crew Intelligence Engine
// ============================================================

model CrewMember {
  id           String     @id @default(uuid())
  productionId String
  production   Production @relation(fields: [productionId], references: [id], onDelete: Cascade)
  name         String
  role         String     // Director / DP / 1st AD / etc.
  department   String     // Camera / Art / Grip / Electric / etc.
  union        String?    // SAG / IATSE / DGA / Teamsters / WGA / Non-Union
  dailyRate    Float?
  weeklyRate   Float?
  startDate    DateTime?
  endDate      DateTime?
  status       String     @default("active") // active / released / hold
  notes        String?
  createdAt    DateTime   @default(now())
  updatedAt    DateTime   @updatedAt
  @@index([productionId])
  @@index([department])
  @@map("crew_members")
}

// ============================================================
// LOCATION NODE
// Connects to scenes and schedule days
// ============================================================

model ProductionLocation {
  id            String     @id @default(uuid())
  productionId  String
  production    Production @relation(fields: [productionId], references: [id], onDelete: Cascade)
  name          String
  address       String?
  city          String?
  state         String?
  country       String?
  locationType  String     // stage / practical / exterior / studio / backlot
  permitStatus  String?    // pending / approved / denied
  dailyCost     Float?
  notes         String?
  createdAt     DateTime   @default(now())
  updatedAt     DateTime   @updatedAt
  @@index([productionId])
  @@map("production_locations")
}

// ============================================================
// BUDGET LINE NODE
// ATL / BTL / POST breakdown. Connects to BudgetRisk engine.
// ============================================================

model BudgetLine {
  id           String     @id @default(uuid())
  productionId String
  production   Production @relation(fields: [productionId], references: [id], onDelete: Cascade)
  category     String     // ATL / BTL / POST / OTHER
  lineItem     String     // e.g. "Director Fee" / "Camera Package"
  accountCode  String?    // industry-standard account codes
  estimated    Float
  actual       Float?
  variance     Float?     // actual - estimated (auto-calculated)
  notes        String?
  createdAt    DateTime   @default(now())
  updatedAt    DateTime   @updatedAt
  @@index([productionId])
  @@index([category])
  @@map("budget_lines")
}

// ============================================================
// PRODUCTION SIGNAL NODE
// The intelligence layer. Any agent writes signals here.
// Dashboard reads them. This is where the OS becomes autonomous.
// ============================================================

model ProductionSignal {
  id           String     @id @default(uuid())
  productionId String
  production   Production @relation(fields: [productionId], references: [id], onDelete: Cascade)
  signalType   String     // budget_drift / ot_spike / weather_risk / schedule_slip / vfx_inflation / crew_conflict / location_issue
  severity     String     // low / medium / high / critical
  source       String?    // aura / compliance / budget / manual
  entityType   String?    // scene / crew / location / budget_line (which node triggered it)
  entityId     String?    // ID of the triggering entity
  message      String
  resolved     Boolean    @default(false)
  resolvedAt   DateTime?
  resolvedBy   String?
  createdAt    DateTime   @default(now())
  updatedAt    DateTime   @updatedAt
  @@index([productionId])
  @@index([signalType])
  @@index([severity])
  @@index([resolved])
  @@map("production_signals")
}
